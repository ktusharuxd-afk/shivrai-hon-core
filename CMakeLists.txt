cmake_minimum_required(VERSION 3.15)
project(shivrai_node CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2 -Wall -Wextra -pthread")

file(GLOB MODULE_SOURCES "${CMAKE_SOURCE_DIR}/src/shivrai/*/*.cpp")
list(FILTER MODULE_SOURCES EXCLUDE REGEX "wallet/")

add_library(shivrai STATIC ${MODULE_SOURCES})
target_include_directories(shivrai PUBLIC "${CMAKE_SOURCE_DIR}/src")

add_executable(shivrai-node node/main.cpp)
target_link_libraries(shivrai-node PRIVATE shivrai pthread)

enable_testing()
file(GLOB TEST_SOURCES "${CMAKE_SOURCE_DIR}/tests/shivrai/test_*.cpp")
foreach(TEST_SRC ${TEST_SOURCES})
    get_filename_component(TEST_NAME ${TEST_SRC} NAME_WE)
    if(TEST_NAME STREQUAL "test_wallet")
        message(STATUS "Skipping ${TEST_NAME} (needs OpenSSL/secp256k1)")
        continue()
    endif()
    string(SUBSTRING "${TEST_NAME}" 5 -1 MODULE_NAME)
    set(MODULE_CPP "${CMAKE_SOURCE_DIR}/src/shivrai/${MODULE_NAME}/${MODULE_NAME}.cpp")
    if(EXISTS "${MODULE_CPP}")
        add_executable(${TEST_NAME} ${TEST_SRC})
        target_link_libraries(${TEST_NAME} PRIVATE shivrai)
        add_test(NAME ${TEST_NAME} COMMAND ${TEST_NAME})
    else()
        message(STATUS "Skipping ${TEST_NAME} (no module source)")
    endif()
endforeach()
